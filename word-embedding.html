def preprocessing_angka(teks):
    teks = teks.lower()
    teks = re.sub(r"http\S+|www\S+", "", teks)
    teks = re.sub(r"\d+", " ", teks)
    teks = re.sub(r"\s+", " ", teks).strip()
    tokens = [kata for kata in teks.split() if kata not in stop_words]
    return [stemmer.stem(kata) for kata in tokens]

df["tokens_angka"] = df["isi_berita"].apply(preprocessing_angka)

model_angka = Word2Vec(
    sentences=df["tokens_angka"].tolist(),
    vector_size=5,
    window=5,
    min_count=1,
    workers=4,
    sg=1,
    seed=42
)

def document_vector(tokens, model):
    vectors = [model.wv[word] for word in tokens if word in model.wv]
    if not vectors:
        return np.zeros(model.vector_size)
    return np.mean(vectors, axis=0)

vsm_angka = np.array([
    document_vector(tokens, model_angka)
    for tokens in df["tokens_angka"]
])
